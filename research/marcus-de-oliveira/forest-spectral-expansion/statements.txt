MathCommons.ForestSpectralExpansion.weightedSubsetExpansion {V : Type} [DecidableEq V] (xs : List V) (unique : xs.Nodup)
  (complete : ∀ (x : V), x ∈ xs) (law : List (MathCommons.ForestPermutationLaw.VertexPermutation V × Rat))
  (a b : V → MathCommons.ForestGateMagnitude.Gaussian) :
  MathCommons.ForestSpectralExpansion.weightedPairings xs law a b =
    MathCommons.ForestGateSupport.scalarSum MathCommons.ForestGateSupport.gaussianArithmetic
      (List.map
        (fun start =>
          MathCommons.ForestGateSupport.scalarSum MathCommons.ForestGateSupport.gaussianArithmetic
            (List.map
              (fun finish =>
                MathCommons.ForestSpectralExpansion.scale
                  (MathCommons.ForestPathNormalization.sumWith
                    (fun entry =>
                      if MathCommons.ForestPermutationLaw.subsetImage entry.fst start = finish then entry.snd else 0)
                    law)
                  (MathCommons.ForestSpectralExpansion.spectralMonomial xs start finish a b))
              (MathCommons.ForestSpectralExpansion.subsets xs)))
        (MathCommons.ForestSpectralExpansion.subsets xs))

MathCommons.ForestSpectralExpansion.forestWeightedSubsetExpansion {V : Type} [DecidableEq V] {vertices : List V}
  (edges : Nat → MathCommons.ForestTrajectory.Edge vertices)
  (as bs cs ds : Nat → MathCommons.ForestGateMagnitude.Gaussian) (n : Nat)
  (bridges : MathCommons.ForestTrajectory.EveryEdgeIsBridge edges n)
  (unitary : ∀ (k : Nat), k < n → MathCommons.ForestGateMagnitude.UnitaryBlock (as k) (bs k) (cs k) (ds k))
  (a b : V → MathCommons.ForestGateMagnitude.Gaussian) :
  (∀ (entry : MathCommons.ForestPermutationLaw.VertexPermutation V × Rat),
      entry ∈ MathCommons.ForestPermutationLaw.coinLaw edges bs n → 0 ≤ entry.snd) ∧
    MathCommons.ForestPathNormalization.sumWith (fun entry => entry.snd)
          (MathCommons.ForestPermutationLaw.coinLaw edges bs n) =
        1 ∧
      MathCommons.ForestSpectralExpansion.weightedPairings vertices
          (MathCommons.ForestPermutationLaw.coinLaw edges bs n) a b =
        MathCommons.ForestGateSupport.scalarSum MathCommons.ForestGateSupport.gaussianArithmetic
          (List.map
            (fun start =>
              MathCommons.ForestGateSupport.scalarSum MathCommons.ForestGateSupport.gaussianArithmetic
                (List.map
                  (fun finish =>
                    MathCommons.ForestSpectralExpansion.scale
                      (MathCommons.ForestGateMagnitude.normSq
                        (MathCommons.ForestGateSupport.minorDet MathCommons.ForestGateSupport.gaussianArithmetic
                          (MathCommons.ForestProductMinor.matrixProduct vertices
                            (MathCommons.ForestPathProbability.gateSchedule edges as bs cs ds) n)
                          (List.filter finish vertices) (List.filter start vertices)))
                      (MathCommons.ForestSpectralExpansion.spectralMonomial vertices start finish a b))
                  (MathCommons.ForestSpectralExpansion.subsets vertices)))
            (MathCommons.ForestSpectralExpansion.subsets vertices))
